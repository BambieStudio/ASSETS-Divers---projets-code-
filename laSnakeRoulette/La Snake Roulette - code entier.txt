const SECTOR_ANGLE  = 360 / TOTAL_SECTORS;

function spectacleFromSector(sectorIndex) {
  return SPECTACLES[sectorIndex % 4];
}

let currentAngle = 0;
let spinning = false;
let selectedTheme = null;

const wheelImg     = document.getElementById('wheelImg');
const spinBtn      = document.getElementById('spinBtn');
const overlay      = document.getElementById('overlay');
const closeBtn     = document.getElementById('closeBtn');

function setRotation(deg) {
  wheelImg.style.transform = `translateX(-50%) rotate(${deg}deg)`;
}

function easeOut(t) { return 1 - Math.pow(1 - t, 4); }

function spin() {
  if (spinning) return;
  spinning = true;
  spinBtn.disabled = true;

  const targetSector     = Math.floor(Math.random() * TOTAL_SECTORS);
  const targetNormalized = targetSector * SECTOR_ANGLE + SECTOR_ANGLE / 2;
  const currentNorm      = (((-currentAngle) % 360) + 360) % 360;
  let delta = targetNormalized - currentNorm;
  if (delta < 0) delta += 360;

  const fullTurns     = (5 + Math.floor(Math.random() * 5)) * 360;
  const totalRotation = fullTurns + delta;
  const duration      = 4500 + Math.random() * 1500;
  const startAngle    = currentAngle;
  const startTime     = performance.now();

  function animate(now) {
    const elapsed = now - startTime;
    const t = Math.min(elapsed / duration, 1);
    currentAngle = startAngle + totalRotation * easeOut(t);
    setRotation(currentAngle);
    if (t < 1) {
      requestAnimationFrame(animate);
    } else {
      currentAngle = startAngle + totalRotation;
      setRotation(currentAngle);
      spinning = false;
      spinBtn.disabled = false;
      setTimeout(() => showResult(spectacleFromSector(targetSector)), 600);
    }
  }
  requestAnimationFrame(animate);
}

  selectedTheme = spectacle.theme;
  spectacleImg.src = spectacle.img;
  spectacleImg.alt = spectacle.id;
  spectacleImg.style.animation = 'none';
  spectacleImg.offsetHeight;
  spectacleImg.style.animation = 'floatIn 0.6s ease forwards';
  overlay.classList.add('visible');
}

function launchGoldenSparkles() {

  const amount = 45;

  for (let i = 0; i < amount; i++) {

    const sparkle = document.createElement('div');
    sparkle.className = 'gold-sparkle';

    // spawn aléatoire en haut de l’écran (pluie)
    const startX = Math.random() * window.innerWidth;
    const startY = -20 - Math.random() * 100;

    sparkle.style.left = `${startX}px`;
    sparkle.style.top  = `${startY}px`;

    // chute verticale dominante
    const fallDistance = 300 + Math.random() * 500;
    const drift = (Math.random() - 0.5) * 120; // léger vent gauche/droite

    sparkle.style.setProperty('--x', `${drift}px`);
    sparkle.style.setProperty('--y', `${fallDistance}px`);

    // tailles variées mais petites (plus réaliste)
    const size = 3 + Math.random() * 5;
    sparkle.style.width = `${size}px`;
    sparkle.style.height = `${size}px`;

    // durée un peu plus fluide
    sparkle.style.animationDuration =
      `${1800 + Math.random() * 1500}ms`;

    document.body.appendChild(sparkle);

    setTimeout(() => sparkle.remove(), 4000);
  }
}
